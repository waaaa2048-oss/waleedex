import "server-only";
import { createHmac, randomInt, timingSafeEqual } from "node:crypto";
import { and, desc, eq, gt, sql } from "drizzle-orm";
import { db } from "@/db";
import { authAttempts, otpCodes, users } from "@/db/schema";

export type OtpKind = "staff" | "customer";

const OTP_TTL_MS = 10 * 60 * 1000;
const MAX_VERIFY_ATTEMPTS = 5;

function hashCode(kind: OtpKind, identifier: string, code: string) {
  const secret = process.env.OTP_SECRET;
  if (!secret) throw new Error("OTP_SECRET غير مضبوط");
  return createHmac("sha256", secret).update(`${kind}:${identifier}:${code}`).digest("hex");
}

/** يوحّد الجوال السعودي إلى صيغة 05XXXXXXXX أو يرجع null. */
export function normalizePhone(input: string): string | null {
  let digits = input
    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)))
    .replace(/\D/g, "");
  if (digits.startsWith("00966")) digits = digits.slice(5);
  else if (digits.startsWith("966")) digits = digits.slice(3);
  if (digits.length === 9 && digits.startsWith("5")) digits = `0${digits}`;
  return /^05\d{8}$/.test(digits) ? digits : null;
}

/** حد للمحاولات: يرجع false إذا تجاوز الحد خلال النافذة. */
export async function hitLimit(key: string, max: number, windowMs: number): Promise<boolean> {
  const now = new Date();
  const resetAt = new Date(now.getTime() + windowMs);
  const [row] = await db.select().from(authAttempts).where(eq(authAttempts.key, key)).limit(1);
  if (!row || row.resetAt <= now) {
    await db
      .insert(authAttempts)
      .values({ key, count: 1, resetAt })
      .onConflictDoUpdate({ target: authAttempts.key, set: { count: 1, resetAt, updatedAt: now } });
    return true;
  }
  if (row.count >= max) return false;
  await db.update(authAttempts).set({ count: row.count + 1, updatedAt: now }).where(eq(authAttempts.key, key));
  return true;
}

export async function issueOtp(kind: OtpKind, identifier: string, subjectId?: number): Promise<string> {
  const code = String(randomInt(0, 1_000_000)).padStart(6, "0");
  
  // حذف أي رموز سابقة مرتبطة بنفس المعرّف
  await db.delete(otpCodes).where(and(eq(otpCodes.kind, kind), eq(otpCodes.identifier, identifier)));
  
  await db.insert(otpCodes).values({
    kind,
    identifier,
    subjectId: subjectId ?? null,
    codeHash: hashCode(kind, identifier, code),
    expiresAt: new Date(Date.now() + OTP_TTL_MS),
  });
  return code;
}

export async function verifyOtp(kind: OtpKind, identifier: string, code: string): Promise<boolean> {
  const [row] = await db
    .select()
    .from(otpCodes)
    .where(and(eq(otpCodes.kind, kind), eq(otpCodes.identifier, identifier), gt(otpCodes.expiresAt, new Date())))
    .orderBy(desc(otpCodes.id))
    .limit(1);
    
  if (!row || row.attempts >= MAX_VERIFY_ATTEMPTS) return false;

  await db.update(otpCodes).set({ attempts: row.attempts + 1 }).where(eq(otpCodes.id, row.id));

  const expected = Buffer.from(row.codeHash, "hex");
  const actual = Buffer.from(hashCode(kind, identifier, code), "hex");
  if (expected.length !== actual.length || !timingSafeEqual(expected, actual)) return false;

  await db.delete(otpCodes).where(eq(otpCodes.id, row.id));
  return true;
}

export async function findStaffByIdentifier(identifier: string) {
  const value = identifier.trim();
  let condition;
  if (value.includes("@")) {
    condition = eq(sql`lower(${users.email})`, value.toLowerCase());
  } else {
    const phone = normalizePhone(value);
    if (!phone) return null;
    condition = eq(users.phone, phone);
  }
  const [user] = await db
    .select({ id: users.id, email: users.email })
    .from(users)
    .where(condition)
    .limit(1);
  return user ?? null;
}

export async function sendOtpEmail(to: string, code: string) {
  const key = process.env.RESEND_API_KEY;
  if (!key) throw new Error("RESEND_API_KEY غير مضبوط");
  const res = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      from: process.env.RESEND_FROM ?? "onboarding@resend.dev",
      to,
      subject: "رمز تسجيل الدخول",
      html: `<div dir="rtl" style="font-family:Tahoma,Arial,sans-serif"><p>رمز التحقق الخاص بك:</p><p style="font-size:28px;font-weight:bold;letter-spacing:6px">${code}</p><p>ينتهي خلال 10 دقائق. لا تشاركه مع أحد.</p></div>`,
    }),
  });
  if (!res.ok) throw new Error(`Resend ${res.status}: ${await res.text()}`);
}

import { NextResponse } from "next/server";
import { createSession, setSessionCookie } from "@/lib/auth";
import { findStaffByIdentifier, hitLimit, verifyOtp } from "@/lib/otp";

export const dynamic = "force-dynamic";

export async function POST(request: Request) {
  const body = await request.json().catch(() => null);
  const identifier = typeof body?.identifier === "string" ? body.identifier.trim() : "";
  const code = typeof body?.code === "string" ? body.code.trim() : "";
  if (!identifier || !/^\d{6}$/.test(code)) {
    return NextResponse.json({ error: "أدخل الرمز المكوّن من 6 أرقام" }, { status: 400 });
  }

  if (!(await hitLimit(`otp:verify:id:${identifier.toLowerCase()}`, 10, 15 * 60 * 1000))) {
    return NextResponse.json({ error: "محاولات كثيرة، حاول لاحقاً" }, { status: 429 });
  }

  try {
    const user = await findStaffByIdentifier(identifier);
    const valid = user ? await verifyOtp("staff", user.id, code) : false;
    if (!user || !valid) {
      return NextResponse.json({ error: "الرمز غير صحيح أو منتهي" }, { status: 400 });
    }
    const { token, expiresAt } = await createSession(user.id);
    await setSessionCookie(token, expiresAt);
    return NextResponse.json({ ok: true });
  } catch (error) {
    console.error("[otp] verify failed:", error);
    return NextResponse.json({ error: "تعذر التحقق من الرمز" }, { status: 500 });
  }
}

import { NextResponse } from "next/server";
import { findStaffByIdentifier, hitLimit, issueOtp, sendOtpEmail } from "@/lib/otp";

export const dynamic = "force-dynamic";

const GENERIC = {
  ok: true,
  message: "إذا كان الحساب موجوداً أرسلنا رمز التحقق إلى بريده المسجّل.",
};

export async function POST(request: Request) {
  const body = await request.json().catch(() => null);
  const identifier = typeof body?.identifier === "string" ? body.identifier.trim() : "";
  if (!identifier) {
    return NextResponse.json({ error: "أدخل البريد أو رقم الجوال" }, { status: 400 });
  }

  const ip = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? "local";
  const allowed =
    (await hitLimit(`otp:req:ip:${ip}`, 10, 15 * 60 * 1000)) &&
    (await hitLimit(`otp:req:id:${identifier.toLowerCase()}`, 3, 10 * 60 * 1000));
  if (!allowed) {
    return NextResponse.json({ error: "محاولات كثيرة، حاول لاحقاً" }, { status: 429 });
  }

  try {
    const user = await findStaffByIdentifier(identifier);
    if (user) {
      // تمرير الـ identifier مع الـ user.id كمعامل ثالث
      const code = await issueOtp("staff", identifier, user.id);
      await sendOtpEmail(user.email, code);
    }
    return NextResponse.json(GENERIC);
  } catch (error) {
    console.error("[otp] request failed:", error);
    return NextResponse.json({ error: "تعذر إرسال الرمز" }, { status: 500 });
  }
}
